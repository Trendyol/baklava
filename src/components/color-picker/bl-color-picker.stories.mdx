export const EventTemplate = () => {
  const handleChange = (e) => {
    const output = e.target.parentElement.querySelector('.output');
    if (output) output.textContent = e.detail;
  };

    <div style="display: flex; align-items: flex-start; gap: 24px;">
      <bl-color-picker value="#f39c12" @bl-color-change=${handleChange}></bl-color-picker>

      </div>
    </div>
  `;
};

# Color Picker

Color Picker component allows users to select colors through an interactive visual interface. It provides a color area for saturation and brightness selection, a hue slider, and text inputs for precise color values.

## Basic Usage

The simplest usage with default values. Click and drag on the color area to select saturation and brightness. Use the hue slider below to change the base color.

</Canvas>

## Display Formats

Color Picker supports two display formats: **HEX** and **RGB**. Click the format label below the input to toggle between them.

## Keyboard Navigation

The Color Picker supports full keyboard navigation:

| Area | Key | Action |
|------|-----|--------|
| Color Area | `Arrow Left/Right` | Adjust saturation |
| Color Area | `Arrow Up/Down` | Adjust brightness |
| Hue Slider | `Arrow Left/Right` | Adjust hue |
| All | `Shift + Arrow` | Larger step (10x) |
| Inputs | `Tab` | Navigate between inputs |
